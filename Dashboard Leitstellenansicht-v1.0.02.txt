// ==UserScript==
// @name         Dashboard Leitstellenansicht
// @namespace    https://leitstellenspiel.de/bielefeld
// @version      v1.0.02
// @license      Design by Bobelle
// @author       Design by Bobelle
// @description  Leitstellenansicht: Übersicht aller Fahrzeuge nach Gebäuden mit Live-Status
// @updateURL    https://github.com/Bobelle-Homebase/ILS-Bielefeld/raw/refs/heads/main/Dashboard Leitstellenansicht.user.js
// @downloadURL  https://github.com/Bobelle-Homebase/ILS-Bielefeld/raw/refs/heads/main/Dashboard Leitstellenansicht.user.js
// @icon         https://www.leitstellenspiel.de/favicon.ico
// @match        https://www.leitstellenspiel.de/*
// @grant        none
// @run-at       document-idle
// ==/UserScript==
/* global $ */

(function () {
    'use strict';

    // Nur im Hauptfenster der Startseite ausführen
    if (window.top !== window.self) return;
    if (window.location.pathname !== '/') return;

    // ------------------------------------------------------------------
    // Konfiguration
    // ------------------------------------------------------------------
    const POLL_INTERVAL = 60 * 1000; // Sicherheits-Abgleich mit der API (ms)

    const GROUPS = {
    fw:    { name: "Feuerwehr", color: "#DB1111", terms: "feuerwehr fw" },
    rd:    { name: "Rettungsdienst", color: "#F36F22", terms: "rettungsdienst rd" },
    pol:   { name: "Polizei", color: "#2980b9", terms: "polizei pol" },
    thw:   { name: "THW", color: "#3b6ea5", terms: "thw technisches hilfswerk" },
    other: { name: "Sonstige", color: "#8a8f98", terms: "sonstige" }
    };

    // Gebäudetyp-ID (LSS) -> Gruppe. Unbekannte Typen landen in "other".
    const BUILDING_GROUP = {
        0: "fw", 1: "fw", 18: "fw",
        2: "rd", 3: "rd", 5: "rd", 12: "rd", 20: "rd",
        6: "pol", 8: "pol", 11: "pol", 13: "pol", 17: "pol", 19: "pol",
        9: "thw", 10: "thw"
    };

    // ------------------------------------------------------------------
    // Zustand
    // ------------------------------------------------------------------
    let vehicles = []; // Fahrzeuge aus /api/vehicles
    let buildings = new Map(); // id -> Gebäude
    let typeNames = {}; // Fahrzeugtyp-ID -> Kurzname
    let overlay = null;
    let pollTimer = null;

    // ------------------------------------------------------------------
    // Hilfsfunktionen
    // ------------------------------------------------------------------
    const esc = s => String(s ?? "").replace(/[&<>"']/g, c => (
        { "&": "&amp;", "<": "&lt;", ">": "&gt;", '"': "&quot;", "'": "&#39;" }[c]
    ));

    const nowTime = () => new Date().toLocaleTimeString("de-DE", { hour: "2-digit", minute: "2-digit" });

    const getJSON = async url => {
        const r = await fetch(url, { credentials: "same-origin" });
        if (!r.ok) throw new Error(`${url}: HTTP ${r.status}`);
        return r.json();
    };

    const vehicleFms = v => v.fms_real ?? v.fms_show ?? 6;

    const typeLabel = v =>
        v.vehicle_type_caption || typeNames[v.vehicle_type] || `Typ ${v.vehicle_type}`;

    const groupOf = b => GROUPS[BUILDING_GROUP[b.building_type] || "other"];

    // ------------------------------------------------------------------
    // Styles
    // ------------------------------------------------------------------
    const style = document.createElement("style");
    style.textContent = `
        #lsd-overlay { position: fixed; inset: 0; z-index: 99999; background: #12151a; color: #e6e8eb;
            overflow-y: auto; font-family: inherit; display: none; }
        #lsd-overlay.open { display: block; }
        #lsd-header { position: sticky; top: 0; z-index: 2; background: #1a1e25; border-bottom: 1px solid #2b313b;
            padding: 12px 20px; display: flex; gap: 16px; align-items: center; flex-wrap: wrap; }
        #lsd-header h3 { margin: 0; font-size: 20px; color: #fff; }
        #lsd-header .lsd-sub { color: #9aa3af; font-size: 13px; }
        #lsd-search { margin-left: auto; min-width: 260px; padding: 6px 10px; border-radius: 4px;
            border: 1px solid #3a414d; background: #12151a; color: #e6e8eb; }
        #lsd-close { background: #DB1111; color: #fff; border: 0; border-radius: 4px; padding: 6px 14px; cursor: pointer; }
        #lsd-body { padding: 16px 20px 40px; display: grid; gap: 14px;
            grid-template-columns: repeat(auto-fill, minmax(340px, 1fr)); align-items: start; }
        #lsd-body .lsd-empty { grid-column: 1 / -1; text-align: center; color: #9aa3af; padding: 40px; }
        .lsd-card { background: #1a1e25; border: 1px solid #2b313b; border-radius: 6px; overflow: hidden; }
        .lsd-card.hidden { display: none; }
        .lsd-card-head { display: flex; gap: 8px; align-items: center; padding: 8px 10px; background: #20252d;
            border-bottom: 1px solid #2b313b; }
        .lsd-dot { width: 10px; height: 10px; border-radius: 50%; flex: none; }
        .lsd-card-head a { color: #fff; font-weight: 600; text-decoration: none; flex: 1; overflow: hidden;
            text-overflow: ellipsis; white-space: nowrap; }
        .lsd-card-head a:hover { text-decoration: underline; }
        .lsd-count { font-size: 12px; color: #9aa3af; white-space: nowrap; }
        .lsd-vehicles { display: flex; flex-direction: column; }
        .lsd-vehicle { display: flex; align-items: center; gap: 10px; padding: 5px 10px; color: #e6e8eb;
            text-decoration: none; border-bottom: 1px solid #242a33; }
        .lsd-vehicle:last-child { border-bottom: 0; }
        .lsd-vehicle:hover { background: #242a33; color: #fff; text-decoration: none; }
        .lsd-status { width: 26px; height: 22px; line-height: 22px; text-align: center; border-radius: 3px;
            font-weight: 700; font-size: 13px; color: #fff; background: #6b7280; flex: none; }
        .lsd-status.s1 { background: #2e9e3e; } .lsd-status.s2 { background: #1f7a2e; }
        .lsd-status.s3 { background: #e08a00; } .lsd-status.s4 { background: #c46a00; }
        .lsd-status.s5 { background: #c9a800; color: #111; }
        .lsd-status.s6 { background: #4b5563; }
        .lsd-status.s7 { background: #c0392b; } .lsd-status.s8 { background: #8e44ad; }
        .lsd-status.s9 { background: #2980b9; }
        .lsd-name { flex: 1; overflow: hidden; text-overflow: ellipsis; white-space: nowrap; }
        .lsd-type { color: #9aa3af; font-size: 12px; width: 70px; text-align: right; overflow: hidden;
            text-overflow: ellipsis; white-space: nowrap; }
        .lsd-time { color: #9aa3af; font-size: 12px; width: 42px; text-align: right; }
        #lsd-floating-btn { position: fixed; right: 14px; bottom: 14px; z-index: 9999; background: #DB1111;
            color: #fff; border: 0; border-radius: 4px; padding: 8px 14px; font-weight: 600; cursor: pointer;
            box-shadow: 0 2px 8px rgba(0,0,0,.4); }
        #lsd-floating-btn:hover { background: #b80d0d; }
    `;
    document.head.appendChild(style);

    // ------------------------------------------------------------------
    // Rendering
    // ------------------------------------------------------------------
    function renderBody() {
        const body = overlay.querySelector("#lsd-body");

        const byBuilding = new Map();
        for (const v of vehicles) {
            if (!byBuilding.has(v.building_id)) byBuilding.set(v.building_id, []);
            byBuilding.get(v.building_id).push(v);
        }

        const sortedBuildings = [...byBuilding.keys()]
            .map(id => buildings.get(id))
            .filter(Boolean)
            .sort((a, b) => a.caption.localeCompare(b.caption, "de", { sensitivity: "base", numeric: true }));

        if (!sortedBuildings.length) {
            body.innerHTML = `<div class="lsd-empty">Keine Fahrzeuge gefunden.</div>`;
            return;
        }

        body.innerHTML = sortedBuildings.map(b => {
            const group = groupOf(b);
            const list = byBuilding.get(b.id)
                .sort((x, y) => x.caption.localeCompare(y.caption, "de", { sensitivity: "base", numeric: true }));
            const search = `${group.terms} ${b.caption}`.toLowerCase();

            return `<div class="lsd-card" id="lsd-building-${b.id}" data-search="${esc(search)}">
                <div class="lsd-card-head">
                    <span class="lsd-dot" style="background:${group.color}" title="${esc(group.name)}"></span>
                    <a href="/buildings/${b.id}" target="_blank">${esc(b.caption)}</a>
                    <span class="lsd-count"></span>
                </div>
                <div class="lsd-vehicles">
                    ${list.map(v => {
                        const fms = vehicleFms(v);
                        return `<a href="/vehicles/${v.id}" target="_blank" class="lsd-vehicle" data-vid="${v.id}">
                            <span class="lsd-status s${fms}">${fms}</span>
                            <span class="lsd-name">${esc(v.caption)}</span>
                            <span class="lsd-type">${esc(typeLabel(v))}</span>
                            <span class="lsd-time" title="Letzter Statuswechsel (seit Öffnen der Ansicht)">–</span>
                        </a>`;
                    }).join("")}
                </div>
            </div>`;
        }).join("");

        body.querySelectorAll(".lsd-card").forEach(updateCount);
        applySearch();
    }

    function updateCount(card) {
        const all = card.querySelectorAll(".lsd-status");
        const free = [...all].filter(s => s.classList.contains("s1") || s.classList.contains("s2")).length;
        card.querySelector(".lsd-count").textContent = `${free}/${all.length} verfügbar`;
        const sub = overlay.querySelector("#lsd-sub");
        if (sub) {
            const total = overlay.querySelectorAll(".lsd-status").length;
            const available = overlay.querySelectorAll(".lsd-status.s1, .lsd-status.s2").length;
            sub.textContent = `${available}/${total} Fahrzeuge verfügbar (S1/S2)`;
        }
    }

    function setVehicleStatus(id, fms, withTime = true) {
        const v = vehicles.find(x => x.id === id);
        if (v) { v.fms_real = fms; }
        if (!overlay) return;
        const el = overlay.querySelector(`.lsd-vehicle[data-vid="${id}"]`);
        if (!el) return;
        const st = el.querySelector(".lsd-status");
        if (st.textContent === String(fms)) return;
        st.className = `lsd-status s${fms}`;
        st.textContent = fms;
        if (withTime) el.querySelector(".lsd-time").textContent = nowTime();
        updateCount(el.closest(".lsd-card"));
    }

    function applySearch() {
        const term = (overlay.querySelector("#lsd-search").value || "").trim().toLowerCase();
        overlay.querySelectorAll(".lsd-card").forEach(card => {
            card.classList.toggle("hidden", !!term && !card.dataset.search.includes(term));
        });
    }

    // ------------------------------------------------------------------
    // Daten laden
    // ------------------------------------------------------------------
    async function loadTypeNames() {
        if (Object.keys(typeNames).length) return;
        try {
            const data = await getJSON("https://api.lss-manager.de/de_DE/vehicles");
            for (const [id, t] of Object.entries(data)) typeNames[id] = t.short_name || t.caption;
        } catch (e) {
            console.warn("[Leitstellenansicht] Fahrzeugtyp-Namen nicht geladen:", e);
        }
    }

    async function loadData() {
        const [v, b] = await Promise.all([
            getJSON("/api/vehicles"),
            getJSON("/api/buildings"),
            loadTypeNames()
        ]);
        vehicles = v;
        buildings = new Map(b.map(x => [x.id, x]));
    }

    async function poll() {
        try {
            const fresh = await getJSON("/api/vehicles");
            for (const v of fresh) {
                const old = vehicles.find(x => x.id === v.id);
                if (old && vehicleFms(old) !== vehicleFms(v)) setVehicleStatus(v.id, vehicleFms(v));
            }
        } catch (e) {
            console.warn("[Leitstellenansicht] Abgleich fehlgeschlagen:", e);
        }
    }

    // ------------------------------------------------------------------
    // Live-Updates über die radioMessage-Funktion des Spiels
    // ------------------------------------------------------------------
    function hookRadioMessage() {
        if (typeof window.radioMessage !== "function" || window.radioMessage.__lsd) return;
        const orig = window.radioMessage;
        const wrapped = function (msg) {
            try {
                if (msg && msg.type === "vehicle_fms" && overlay && overlay.classList.contains("open")) {
                    const fms = msg.fms_real ?? msg.fms;
                    if (msg.id != null && fms != null) setVehicleStatus(Number(msg.id), Number(fms));
                }
            } catch (e) { /* nie das Spiel stören */ }
            return orig.apply(this, arguments);
        };
        wrapped.__lsd = true;
        window.radioMessage = wrapped;
    }

    // ------------------------------------------------------------------
    // Overlay
    // ------------------------------------------------------------------
    function buildOverlay() {
        overlay = document.createElement("div");
        overlay.id = "lsd-overlay";
        overlay.innerHTML = `
            <div id="lsd-header">
                <h3>Leitstellenansicht</h3>
                <span class="lsd-sub" id="lsd-sub">Lade Daten …</span>
                <input type="text" id="lsd-search" placeholder="Suche (Gebäude, FW, RD, Pol …)">
                <button id="lsd-close" type="button">Schließen (Esc)</button>
            </div>
            <div id="lsd-body"><div class="lsd-empty">Lade Daten …</div></div>`;
        document.body.appendChild(overlay);

        overlay.querySelector("#lsd-search").addEventListener("input", applySearch);
        overlay.querySelector("#lsd-close").addEventListener("click", closeOverlay);
        document.addEventListener("keydown", e => {
            if (e.key === "Escape" && overlay.classList.contains("open")) closeOverlay();
        });
    }

    async function openOverlay() {
        if (!overlay) buildOverlay();
        overlay.classList.add("open");
        document.body.style.overflow = "hidden";
        hookRadioMessage();
        try {
            await loadData();
            renderBody();
        } catch (e) {
            console.error("[Leitstellenansicht]", e);
            overlay.querySelector("#lsd-body").innerHTML =
                `<div class="lsd-empty">Daten konnten nicht geladen werden (${esc(e.message)}).</div>`;
        }
        clearInterval(pollTimer);
        pollTimer = setInterval(poll, POLL_INTERVAL);
    }

    function closeOverlay() {
        overlay.classList.remove("open");
        document.body.style.overflow = "";
        clearInterval(pollTimer);
    }

    // ------------------------------------------------------------------
    // Menüeintrag
    // ------------------------------------------------------------------
    function addMenuEntry() {
        const btn = document.createElement("button");
        btn.id = "lsd-floating-btn";
        btn.type = "button";
        btn.textContent = "Leitstellenansicht";
        btn.addEventListener("click", openOverlay);
        document.body.appendChild(btn);
    }

    addMenuEntry();
})();
